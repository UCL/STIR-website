93template <typename elemT>
94__device__ inline void
95blockReduction(elemT* shared_mem, int thread_in_block, int block_threads)
96{
97 for (int stride = block_threads / 2; stride > 0; stride /= 2)
98 {
99 if (thread_in_block < stride)
100 shared_mem[thread_in_block] += shared_mem[thread_in_block + stride];
101 __syncthreads();
102 }
103}
