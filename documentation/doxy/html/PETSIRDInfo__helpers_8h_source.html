131inline bool
132get_spacing_uniform(std::vector<float>& spacing, const std::set<float>& unsorted_block_poss, double epsilon = 1e-4)
133{
134 std::vector<float> sorted_z(unsorted_block_poss.begin(), unsorted_block_poss.end());
135 for (size_t i = 1; i < sorted_z.size(); ++i)
136 {
137 spacing.push_back(std::abs(sorted_z[i] - sorted_z[i - 1]));
138 }
139
140 return std::all_of(spacing.begin(), spacing.end(), [&](float s) { return std::abs(s - spacing.front()) <= epsilon; });
141}

204// Detect groupSize along dim2 (y) and loops along dim3 (z)
205// Returns true on success, false if pattern doesn't match the assumed structure.
206bool
207inferGroupSizes_dim2_dim3(const std::vector<stir::CartesianCoordinate3D<float>>& pts,
208 std::size_t& groupSize_dim2,
209 std::size_t& groupSize_dim3,
210 float tol = 1e-5f)
211{
212 const std::size_t n = pts.size();
213 groupSize_dim2 = groupSize_dim3 = 0;
214
215 if (n == 0)
216 return false;
217
218 if (n == 1)
219 {
220 groupSize_dim2 = 1;
221 groupSize_dim3 = 1;
222 return true;
223 }
224
225 // STIR CartesianCoordinate3D<T> is (z, y, x)
226 const float x0 = pts[0].x();
227 const float z0 = pts[0].z();
228
229 // 1) Find how many initial points keep x and z the same
230 std::size_t runLen = 1;
231 while (runLen < n && almostEqual(pts[runLen].x(), x0, tol) && almostEqual(pts[runLen].z(), z0, tol))
232 {
233 ++runLen;
234 }
235
236 groupSize_dim2 = runLen;
237
238 // Must tile the full array
239 if (groupSize_dim2 == 0 || n % groupSize_dim2 != 0)
240 return false;
241
242 groupSize_dim3 = n / groupSize_dim2;
243
244 // 2) Check each block of size groupSize_dim2 has constant x,z
245 for (std::size_t b = 0; b < groupSize_dim3; ++b)
246 {
247 std::size_t start = b * groupSize_dim2;
248 float xb = pts[start].x();
249 float zb = pts[start].z();
250
251 for (std::size_t i = 1; i < groupSize_dim2; ++i)
252 {
253 const auto& p = pts[start + i];
254 if (!almostEqual(p.x(), xb, tol) || !almostEqual(p.z(), zb, tol))
255 {
256 return false; // pattern breaks inside a block
257 }
258 }
259 }
260
261 // 3) Optionally check that z actually changes between blocks
262 for (std::size_t b = 1; b < groupSize_dim3; ++b)
263 {
264 float z_prev = pts[(b - 1) * groupSize_dim2].z();
265 float z_curr = pts[b * groupSize_dim2].z();
266 if (almostEqual(z_prev, z_curr, tol))
267 {
268 return false; // outer loop didn't move in z
269 }
270 }
271
272 return true;
273}
